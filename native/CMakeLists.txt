cmake_minimum_required(VERSION 3.20)
project(system1_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT LLAMA_SOURCE_DIR OR NOT LLAMA_BUILD_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR and LLAMA_BUILD_DIR to the pinned llama.cpp source/build")
endif()
find_library(LLAMA_LIBRARY NAMES llama PATHS "${LLAMA_BUILD_DIR}/bin" "${LLAMA_BUILD_DIR}/src" "${LLAMA_BUILD_DIR}/src/Release" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_LIBRARY NAMES ggml PATHS "${LLAMA_BUILD_DIR}/bin" "${LLAMA_BUILD_DIR}/ggml/src" "${LLAMA_BUILD_DIR}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
find_library(GGML_BASE_LIBRARY NAMES ggml-base PATHS "${LLAMA_BUILD_DIR}/bin" "${LLAMA_BUILD_DIR}/ggml/src" "${LLAMA_BUILD_DIR}/ggml/src/Release" NO_DEFAULT_PATH REQUIRED)
add_executable(system1-native engine.cpp)
target_include_directories(system1-native PRIVATE "${LLAMA_SOURCE_DIR}/include" "${LLAMA_SOURCE_DIR}/ggml/include" "${LLAMA_SOURCE_DIR}/vendor")
target_link_libraries(system1-native PRIVATE "${LLAMA_LIBRARY}" "${GGML_LIBRARY}" "${GGML_BASE_LIBRARY}")
set_target_properties(system1-native PROPERTIES BUILD_RPATH "$ORIGIN" RUNTIME_OUTPUT_DIRECTORY "${LLAMA_BUILD_DIR}/bin")

if(WIN32)
  set_target_properties(system1-native PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${LLAMA_BUILD_DIR}/bin/Release")
endif()
